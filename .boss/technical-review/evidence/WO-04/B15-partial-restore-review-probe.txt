function validateSDKActorRequest(request, mode) {
  const rejected = () => { throw new Error('Invalid private SDK request.'); };
  if (!request || request.version !== 1 || request.kind !== (mode === 'replay' ? 'count-sdk-replay' : 'count-sdk-maintenance') ||
      !/^demo-[a-z0-9-]{1,58}$/.test(request.projectId ?? '') || !countId(request.owner) || !countHash(request.expectedManifestHash) ||
      request.manifest?.version !== 1 || request.manifest?.kind !== 'comment-count-plan' || request.manifest.projectId !== request.projectId || request.manifest.root !== request.root || request.manifest.hash !== request.expectedManifestHash ||
      typeof request.directory !== 'string' || !isAbsolute(request.directory) || !Number.isSafeInteger(request.expiresAt) ||
      !Number.isInteger(request.maxItems) || request.maxItems < 1 || request.maxItems > 25 ||
      !Array.isArray(request.keyBytes) || request.keyBytes.length !== 32 || request.keyBytes.some(byte => !Number.isInteger(byte) || byte < 0 || byte > 255) ||
      (request.nowMs !== undefined && (!Number.isSafeInteger(request.nowMs) || request.nowMs <= 0)) ||
      (mode === 'maintenance' && !['apply', 'restore'].includes(request.operation))) rejected();
  countRoot(request.root);
  const host = /^(127\.0\.0\.1|localhost|\[::1\]):(\d+)$/.exec(request.emulatorHost ?? '');
  if (!host || Number(host[2]) < 1 || Number(host[2]) > 65535 || (process.env.FIRESTORE_EMULATOR_HOST && process.env.FIRESTORE_EMULATOR_HOST !== request.emulatorHost)) rejected();
  const now = request.nowMs === undefined ? Date.now : () => request.nowMs;
  if (request.expiresAt - now() < 3600000 || request.expiresAt > now() + 30 * 86400000) rejected();
  return { operation: mode === 'replay' ? 'apply' : request.operation, now };
}

async function authenticatedReviewReplay(f) {
  const name = (await fsPromises.readdir(f.config.directory)).find(value => value.endsWith('.counts-replay'));
  assert.ok(name, 'The actual successful apply must have published its genuine private replay head.');
  const path = join(f.config.directory, name), envelope = JSON.parse(await readFile(path, 'utf8'));
  const identity = await fsPromises.lstat(f.config.directory);
  const aad = Buffer.from(`comment-count-replay:1:${envelope.bindingHash}:${identity.dev}:${identity.ino}:${envelope.expiresAt}`);
  const decipher = createDecipheriv('aes-256-gcm', f.config.key, Buffer.from(envelope.nonce, 'base64'));
  decipher.setAAD(aad); decipher.setAuthTag(Buffer.from(envelope.tag, 'base64'));
  const value = JSON.parse(Buffer.concat([decipher.update(Buffer.from(envelope.ciphertext, 'base64')), decipher.final()]).toString('utf8'));
  return { path, envelope, aad, value };
}
async function publishReviewReplayType(f, original, field, value) {
  const body = { ...original.value, [field]: value }, nonce = randomBytes(12);
  const cipher = createCipheriv('aes-256-gcm', f.config.key, nonce); cipher.setAAD(original.aad);
  const ciphertext = Buffer.concat([cipher.update(JSON.stringify(body)), cipher.final()]);
  await writeFile(original.path, JSON.stringify({ ...original.envelope,
    hash: createHash('sha256').update(JSON.stringify(encodeCleanupValue(body))).digest('hex'),
    nonce: nonce.toString('base64'), tag: cipher.getAuthTag().toString('base64'), ciphertext: ciphertext.toString('base64') }));
}
